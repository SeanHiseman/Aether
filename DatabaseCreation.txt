CREATE TABLE users (
    user_id CHAR(36) PRIMARY KEY,
    username VARCHAR(120) NOT NULL,
    password VARCHAR(120) NOT NULL,
    user_since TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    collaborative_preference FLOAT NOT NULL DEFAULT 0.5,
    time_preference FLOAT NOT NULL DEFAULT 0.0001,
    has_membership BOOLEAN DEFAULT FALSE,
    theme VARCHAR(120) DEFAULT 'dark',
    points INT DEFAULT 0
);

CREATE TABLE feeds (
    feed_id CHAR(36) PRIMARY KEY,
    parent_id CHAR(36),
    feed_name VARCHAR(100) NOT NULL,
    description VARCHAR(1000),
    feed_photo TEXT,
    follower_count INT DEFAULT 0,
    date_created TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    type VARCHAR(10) DEFAULT 'public',
    is_group BOOLEAN DEFAULT FALSE,
    feed_owner CHAR(36) NOT NULL,
    FOREIGN KEY (feed_owner) REFERENCES users(user_id)
);

CREATE TABLE feed_channels (
    channel_id CHAR(36) PRIMARY KEY,
    channel_name VARCHAR(100) NOT NULL,
    feed_id CHAR(36) NOT NULL,
    is_posts BOOLEAN DEFAULT TRUE,
    is_chat BOOLEAN DEFAULT TRUE,
    date_created TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (feed_id) REFERENCES feeds(feed_id)
);

CREATE TABLE feed_channel_messages (
    message_id CHAR(36) PRIMARY KEY,
    message_content VARCHAR(1000) NOT NULL,
    feed_id CHAR(36) NOT NULL,
    channel_id CHAR(36) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    sender_id CHAR(36) NOT NULL,
    FOREIGN KEY (feed_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (channel_id) REFERENCES feed_channels(channel_id)
);

CREATE TABLE followers (
    follow_id CHAR(36) PRIMARY KEY,
    follower_id CHAR(36) NOT NULL,
    feed_id CHAR(36) NOT NULL,
    is_mod BOOLEAN DEFAULT FALSE,
    is_admin BOOLEAN DEFAULT FALSE,
    follow_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (follower_id) REFERENCES users(user_id),
    FOREIGN KEY (feed_id) REFERENCES feeds(feed_id)
);

CREATE TABLE follow_requests (
    request_id CHAR(36) PRIMARY KEY,
    sender_id CHAR(36) NOT NULL,
    receiver_id CHAR(36) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (receiver_id) REFERENCES feeds(feed_id)
);

CREATE TABLE nested_feed (
    sub_feed_id CHAR(36),
    parent_feed_id CHAR(36),
    PRIMARY KEY (sub_feed_id, parent_feed_id),
    FOREIGN KEY (sub_feed_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (parent_feed_id) REFERENCES feeds(feed_id)
);

CREATE TABLE posts (
    post_id CHAR(36) PRIMARY KEY,
    parent_id CHAR(36),
    feed_id CHAR(36) NOT NULL,
    channel_id CHAR(36) NOT NULL,
    title VARCHAR(120),
    content TEXT NOT NULL,
    replies INT DEFAULT 0,
    views INT DEFAULT 0,
    upvotes INT DEFAULT 0,
    downvotes INT DEFAULT 0,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    poster_id CHAR(36) NOT NULL,
    points INT DEFAULT 0,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (feed_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (channel_id) REFERENCES feed_channels(channel_id)
);

CREATE TABLE post_notes (
    note_id CHAR(36) PRIMARY KEY,
    post_id CHAR(36) NOT NULL,
    note_content VARCHAR(1000) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_misinfo BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (post_id) REFERENCES posts(post_id)
);

CREATE TABLE post_votes (
    vote_id CHAR(36) PRIMARY KEY,
    content_id CHAR(36) NOT NULL,
    voter_id CHAR(36) NOT NULL,
    vote_count INT DEFAULT 0
);

CREATE TABLE ask_chats (
    chat_id CHAR(36) PRIMARY KEY,
    name VARCHAR(256) DEFAULT 'New chat',
    user_id CHAR(36) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);

CREATE TABLE ask_messages (
    message_id CHAR(36) PRIMARY KEY,
    chat_id CHAR(36) NOT NULL,
    sender_id CHAR(36) NOT NULL,
    message_content VARCHAR(1000) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (chat_id) REFERENCES ask_chats(chat_id),
    FOREIGN KEY (sender_id) REFERENCES users(user_id)
);

CREATE TABLE chats (
    chat_id CHAR(36) PRIMARY KEY,
    title VARCHAR(256) NOT NULL DEFAULT 'New chat',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE connections (
    connection_id CHAR(36) PRIMARY KEY,
    feed1_id CHAR(36) NOT NULL,
    feed2_id CHAR(36) NOT NULL,
    connection_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (feed1_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (feed2_id) REFERENCES feeds(feed_id)
);

CREATE TABLE connect_requests (
    request_id CHAR(36) PRIMARY KEY,
    sender_id CHAR(36) NOT NULL,
    receiver_id CHAR(36) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (receiver_id) REFERENCES feeds(feed_id)
);

CREATE TABLE feed_chats (
    feed_id CHAR(36),
    chat_id CHAR(36),
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (feed_id, chat_id),
    FOREIGN KEY (feed_id) REFERENCES feeds(feed_id),
    FOREIGN KEY (chat_id) REFERENCES chats(chat_id)
);

CREATE TABLE messages (
    message_id CHAR(36) PRIMARY KEY,
    chat_id CHAR(36) NOT NULL,
    sender_id CHAR(36) NOT NULL,
    message_content VARCHAR(1000) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (chat_id) REFERENCES chats(chat_id),
    FOREIGN KEY (sender_id) REFERENCES users(user_id)
);
