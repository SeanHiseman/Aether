	const formatSentiment = (value) => {
		if (value < -0.3) return 'Negative';
		if (value > 0.3) return 'Positive';
		return 'Neutral';
	};

	const formatContentVariety = (value) => {
		if (value < 0.3) return 'Random';
		if (value > 0.7) return 'Similar';
		return 'Mixed';
	};

	const formatVoteImpact = (value) => {
		if (value < -0.3) return 'None';
		if (value > 0.3) return 'Heavy';
		return 'Mild';
	};

	const formatContentTypes = (types) => {
		const enabled = Object.entries(types).filter(([_, value]) => value).map(([key, _]) => key);
		return enabled.length > 0 ? enabled.join(', ') : 'None selected';
	};

	const formatTemplate = (template) => {
		const templateMap = {
			'none': 'None',
			'work': 'Work Focus',
			'weekend': 'Weekend Leisure',
			'news': 'News Only'
		};
		return templateMap[template] || 'None';
	};

	const formatChronology = (chronology) => {
		const chronologyMap = {
			'newest': 'Newest First',
			'oldest': 'Oldest First',
			'mixed': 'Mixed Chronology'
		};
		return chronologyMap[chronology] || 'Newest First';
	};

	const generateSummary = () => {
		const boostWords = wordBoost.split(',').map(w => w.trim()).filter(Boolean);
		const suppressWords = wordSuppress.split(',').map(w => w.trim()).filter(Boolean);
		
		return [
			`Name: ${algorithmName || 'Unnamed Algorithm'}`,
			`Template: ${formatTemplate(template)}`,
			`Chronology: ${formatChronology(chronology)}`,
			`Sentiment: ${formatSentiment(sentiment)} (${sentiment})`,
			`Content Variety: ${formatContentVariety(strength)} (${strength})`,
			`Vote Impact: ${formatVoteImpact(similarity)} (${similarity})`,
			`Content Types: ${formatContentTypes(contentType)}`,
			`Active Hours: ${startTime} - ${endTime}`,
			`Boost Words: ${boostWords.length > 0 ? boostWords.join(', ') : 'None'}`,
			`Suppress Words: ${suppressWords.length > 0 ? suppressWords.join(', ') : 'None'}`,
			`Custom Instruction: ${customInstruction || 'None'}`
		];
	};



    <div className="algorithm-summary">
        <p className="small-text">Algorithm Summary</p>
        <div className="summary-list">
            {generateSummary().map((item, index) => (
                <div key={index} className="tiny-text">
                    {item}
                </div>
            ))}
        </div>
    </div>

	<div className="form-row">
		<div className="form-group">
			<div className="form-label-with-info">
				<label className="small-text">Active Days</label>
				<div className="info-icon" title="Select which days of the week this algorithm should be active.">
					<FaInfoCircle />
				</div>
			</div>
			<div>
				{Object.entries(activeDays).map(([day, isActive]) => (
					<label key={day} style={{ marginRight: '1rem' }}>
						<input
							type="checkbox"
							checked={isActive}
							onChange={e => setActiveDays(prev => ({ ...prev, [day]: e.target.checked }))}
							style={{ accentColor: '#007bff' }}
						/>
						{day.charAt(0).toUpperCase() + day.slice(1, 3)}
					</label>
				))}
			</div>
		</div>
	</div>
