    channel_id,
    channel_name,
    created_at,
    display_order,
    saver_id,
    updated_at

    UUID(),
    'Main',
    CURRENT_TIMESTAMP(3),
    0,
    f.feed_id,
    CURRENT_TIMESTAMP(3)

    feeds AS f
WHERE
    f.is_group = FALSE
  AND NOT EXISTS (
    SELECT 1
    FROM saved_post_channels AS spc
